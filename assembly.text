# Register setup stuff
li s0, -4              #LED address stored (store -4 in reg s0)
li s1, 0x00FF0000      #Red stored 00000000 11111111 00000000 00000000
li s2, 0x0000FF00      # Green stored
li s3, 0x000000FF      # Blue stored 
li s4, 500             # Delay (ms) stored in s4

loop:
    # RED LD
    sw s1, 0(s0)        #red and -4 (0 is offset) = -4
    lw t0, -12(zero)       # time start

red_hold:
    lw t1, -12(zero)      # time current 
    sub t2, t1, t0         # subtract 
    bltu t2, s4, red_hold  #branch if less than(compare)

    # GREEN LED
    sw s2, 0(s0)
    lw t0, -12(zero)

green_hold:
    lw t1, -12(zero)
    sub t2, t1, t0
    bltu t2, s4, green_hold

    # BLUE LED
    sw s3, 0(s0)
    lw t0, -12(zero)

blue_hold:
    lw t1, -12(zero)
    sub t2, t1, t0
    bltu t2, s4, blue_hold

    # Repeat.. returns to loop
    j loop
